# User Auth + Session Isolation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add application-level user authentication (register/login/logout) with bcrypt passwords and HttpOnly cookie sessions, isolating conversations per user.

**Architecture:** Go backend adds `poc_users` + `poc_sessions` tables, auth middleware intercepts `/api/*` (excluding `/api/auth/*`), injects user_id into request context. Frontend adds LoginPage component, App.tsx gates on auth state. Conversations are filtered by user_id.

**Tech Stack:** Go 1.25 + golang.org/x/crypto/bcrypt (already in go.mod via excelize), React + TypeScript

import (

	"regexp"
	"time"

var validUsername = regexp.MustCompile(`^[a-zA-Z0-9_]{3,32}$`)

type AuthService struct {
	db *sql.DB
}

func NewAuthService(db *sql.DB) (*AuthService, error) {
	const usersDDL = `CREATE TABLE IF NOT EXISTS poc_users (
		id         VARCHAR(64) PRIMARY KEY,
		username   VARCHAR(32) UNIQUE NOT NULL,
		password   VARCHAR(128) NOT NULL,
		created_at DATETIME DEFAULT NOW()
	)`
	if _, err := db.Exec(usersDDL); err != nil {
		return nil, fmt.Errorf("create poc_users: %w", err)
	}

	const sessionsDDL = `CREATE TABLE IF NOT EXISTS poc_sessions (
		token      VARCHAR(128) PRIMARY KEY,
		user_id    VARCHAR(64) NOT NULL,
		expires_at DATETIME NOT NULL
	)`
	if _, err := db.Exec(sessionsDDL); err != nil {
		return nil, fmt.Errorf("create poc_sessions: %w", err)
	}

	svc := &AuthService{db: db}
	go svc.cleanupLoop()
	return svc, nil
}

func (s *AuthService) Register(username, password string) error {
	if !validUsername.MatchString(username) {
		return fmt.Errorf("username must be 3-32 characters [a-zA-Z0-9_]")
	}
	if len(password) < 6 {
		return fmt.Errorf("password must be at least 6 characters")
	}

	hash, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	if err != nil {
		return fmt.Errorf("hash password: %w", err)
	}

	id := newUUID()
	_, err = s.db.Exec(
		`INSERT INTO poc_users (id, username, password) VALUES (?, ?, ?)`,
		id, username, string(hash),
	)
	if err != nil {
		if isDuplicateEntry(err) {
			return fmt.Errorf("username already taken")
		}
		return fmt.Errorf("create user: %w", err)
	}
	log.Printf("auth: registered user %s (%s)", username, id)
	return nil
}

func isDuplicateEntry(err error) bool {
	return err != nil && (contains(err.Error(), "Duplicate entry") || contains(err.Error(), "duplicate key"))
}

func contains(s, sub string) bool {
	return len(s) >= len(sub) && searchString(s, sub)
}

func searchString(s, sub string) bool {
	for i := 0; i <= len(s)-len(sub); i++ {
		if s[i:i+len(sub)] == sub {
			return true
		}
	}
	return false
}
```

func (s *AuthService) Login(username, password string) (string, error) {
	var userID, hash string
	err := s.db.QueryRow(
		`SELECT id, password FROM poc_users WHERE username = ?`, username,
	).Scan(&userID, &hash)
	if err == sql.ErrNoRows {
		return "", fmt.Errorf("invalid username or password")
	}
	if err != nil {
		return "", fmt.Errorf("query user: %w", err)
	}

	if err := bcrypt.CompareHashAndPassword([]byte(hash), []byte(password)); err != nil {
		return "", fmt.Errorf("invalid username or password")
	}

	token, err := generateToken()
	if err != nil {
		return "", fmt.Errorf("generate token: %w", err)
	}

	expiresAt := time.Now().Add(7 * 24 * time.Hour)
	_, err = s.db.Exec(
		`INSERT INTO poc_sessions (token, user_id, expires_at) VALUES (?, ?, ?)`,
		token, userID, expiresAt,
	)
	if err != nil {
		return "", fmt.Errorf("create session: %w", err)
	}

	log.Printf("auth: user %s logged in", username)
	return token, nil
}

func generateToken() (string, error) {
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return hex.EncodeToString(b), nil
}

func (s *AuthService) ValidateSession(token string) (string, bool) {
	var userID string
	err := s.db.QueryRow(
		`SELECT user_id FROM poc_sessions WHERE token = ? AND expires_at > NOW()`, token,
	).Scan(&userID)
	if err != nil {
		return "", false
	}
	return userID, true
}

func (s *AuthService) GetUsername(userID string) (string, error) {
	var username string
	err := s.db.QueryRow(`SELECT username FROM poc_users WHERE id = ?`, userID).Scan(&username)
	if err != nil {
		return "", err
	}
	return username, nil
}

func (s *AuthService) Logout(token string) {
	s.db.Exec(`DELETE FROM poc_sessions WHERE token = ?`, token)
}

func (s *AuthService) cleanupLoop() {
	ticker := time.NewTicker(1 * time.Hour)
	defer ticker.Stop()
	for range ticker.C {
		res, _ := s.db.Exec(`DELETE FROM poc_sessions WHERE expires_at <= NOW()`)
		if n, _ := res.RowsAffected(); n > 0 {
			log.Printf("auth: cleaned up %d expired sessions", n)
		}
	}
}

const cookieName = "poc_token"
const cookieMaxAge = 7 * 24 * 60 * 60 // 7 days in seconds

func setTokenCookie(w http.ResponseWriter, token string) {
	http.SetCookie(w, &http.Cookie{
		Name:     cookieName,
		Value:    token,
		Path:     "/",
		HttpOnly: true,
		SameSite: http.SameSiteLaxMode,
		MaxAge:   cookieMaxAge,
	})
}

func clearTokenCookie(w http.ResponseWriter) {
	http.SetCookie(w, &http.Cookie{
		Name:     cookieName,
		Value:    "",
		Path:     "/",
		HttpOnly: true,
		SameSite: http.SameSiteLaxMode,
		MaxAge:   -1,
	})
}

package main

import (
	"context"
	"encoding/json"
	"net/http"
	"strings"
)

type contextKey string

const userIDKey contextKey = "user_id"

func UserIDFromContext(ctx context.Context) string {
	if v, ok := ctx.Value(userIDKey).(string); ok {
		return v
	}
	return ""
}

func authMiddleware(auth *AuthService, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		path := r.URL.Path

		if strings.HasPrefix(path, "/api/auth/") || strings.HasPrefix(path, "/api/auth") {
			next.ServeHTTP(w, r)
			return
		}

		if !strings.HasPrefix(path, "/api/") {
			next.ServeHTTP(w, r)
			return
		}

		if r.Method == http.MethodOptions {
			next.ServeHTTP(w, r)
			return
		}

		cookie, err := r.Cookie(cookieName)
		if err != nil || cookie.Value == "" {
			writeUnauthorized(w)
			return
		}

		userID, ok := auth.ValidateSession(cookie.Value)
		if !ok {
			writeUnauthorized(w)
			return
		}

		ctx := context.WithValue(r.Context(), userIDKey, userID)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}

func writeUnauthorized(w http.ResponseWriter) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusUnauthorized)
	json.NewEncoder(w).Encode(map[string]string{"error": "unauthorized"})
}
```

```typescript
const BASE = '/api/auth'

export interface AuthUser {
  username: string
}

async function parseJSON<T>(resp: Response): Promise<T> {
  const data = await resp.json()
  if (!resp.ok) {
    throw new Error(data.error || `HTTP ${resp.status}`)
  }
  return data as T
}

export async function login(username: string, password: string): Promise<AuthUser> {
  const resp = await fetch(`${BASE}/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ username, password }),
  })
  return parseJSON<AuthUser>(resp)
}

export async function register(username: string, password: string): Promise<AuthUser> {
  const resp = await fetch(`${BASE}/register`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ username, password }),
  })
  return parseJSON<AuthUser>(resp)
}

export async function logout(): Promise<void> {
  await fetch(`${BASE}/logout`, { method: 'POST' })
}

export async function getMe(): Promise<AuthUser | null> {
  try {
    const resp = await fetch(`${BASE}/me`)
    if (resp.status === 401) return null
    return parseJSON<AuthUser>(resp)
  } catch {
    return null
  }
}
```

## Task 6: LoginPage component

**Files:**
- Create: `web/src/components/LoginPage.tsx`

- [ ] **Step 1: Create LoginPage with login/register tabs**

```tsx
import { useState } from 'react'
import { useT, tpl } from '../i18n'
import { login, register } from '../api/auth'

interface Props {
  onLogin: (username: string) => void
}

export function LoginPage({ onLogin }: Props) {
  const { t } = useT()
  const [mode, setMode] = useState<'login' | 'register'>('login')
  const [username, setUsername] = useState('')
  const [password, setPassword] = useState('')
  const [confirm, setConfirm] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setError('')

    if (mode === 'register' && password !== confirm) {
      setError(t('passwordMismatch'))
      return
    }

    setLoading(true)
    try {
      if (mode === 'login') {
        const user = await login(username, password)
        onLogin(user.username)
      } else {
        const user = await register(username, password)
        onLogin(user.username)
      }
    } catch (err: any) {
      setError(err.message || (mode === 'login' ? t('loginError') : t('registerError')))
    } finally {
      setLoading(false)
    }
  }

  const cardStyle: React.CSSProperties = {
    maxWidth: 380,
    margin: '120px auto',
    padding: '32px 28px',
    background: '#fff',
    borderRadius: 12,
    boxShadow: '0 4px 24px rgba(0,0,0,0.08)',
  }

  const tabRow: React.CSSProperties = {
    display: 'flex',
    gap: 0,
    marginBottom: 24,
    borderBottom: '2px solid #f0f0f0',
  }

  const tabStyle = (active: boolean): React.CSSProperties => ({
    flex: 1,
    padding: '10px 0',
    textAlign: 'center',
    cursor: 'pointer',
    fontWeight: active ? 600 : 400,
    color: active ? '#1a1a2e' : '#9ca3af',
    borderBottom: active ? '2px solid #1a1a2e' : '2px solid transparent',
    marginBottom: -2,
    background: 'none',
    border: 'none',
    fontSize: 15,
  })

  const inputStyle: React.CSSProperties = {
    width: '100%',
    padding: '10px 12px',
    border: '1px solid #d0d5dd',
    borderRadius: 6,
    fontSize: 14,
    outline: 'none',
    boxSizing: 'border-box',
  }

  const btnStyle: React.CSSProperties = {
    width: '100%',
    padding: '10px 0',
    background: '#1a1a2e',
    color: '#fff',
    border: 'none',
    borderRadius: 6,
    fontSize: 15,
    fontWeight: 600,
    cursor: loading ? 'not-allowed' : 'pointer',
    opacity: loading ? 0.7 : 1,
  }

  return (
    <div style={{ minHeight: '100vh', background: '#f5f6fa' }}>
      <div style={{ textAlign: 'center', paddingTop: 60 }}>
        <div style={{ display: 'inline-flex', alignItems: 'center', gap: 10 }}>
          <div style={{
            width: 40, height: 40, borderRadius: 8,
            background: 'linear-gradient(135deg, #0f3460, #1a1a2e)',
            color: '#fff', fontSize: 16, fontWeight: 700,
            display: 'flex', alignItems: 'center', justifyContent: 'center',
          }}>HK</div>
          <span style={{ fontSize: 20, fontWeight: 600, color: '#1a1a2e' }}>
            {t('title')}
          </span>
        </div>
      </div>

      <div style={cardStyle}>
        <div style={tabRow}>
          <button style={tabStyle(mode === 'login')} onClick={() => { setMode('login'); setError('') }}>
            {t('login')}
          </button>
          <button style={tabStyle(mode === 'register')} onClick={() => { setMode('register'); setError('') }}>
            {t('register')}
          </button>
        </div>

        <form onSubmit={handleSubmit}>
          <div style={{ marginBottom: 16 }}>
            <label style={{ display: 'block', fontSize: 13, color: '#475467', marginBottom: 6 }}>
              {t('username')}
            </label>
            <input
              style={inputStyle}
              value={username}
              onChange={e => setUsername(e.target.value)}
              autoComplete="username"
            />
            {mode === 'register' && (
              <div style={{ fontSize: 11, color: '#9ca3af', marginTop: 4 }}>{t('usernameHint')}</div>
            )}
          </div>

          <div style={{ marginBottom: 16 }}>
            <label style={{ display: 'block', fontSize: 13, color: '#475467', marginBottom: 6 }}>
              {t('password')}
            </label>
            <input
              style={inputStyle}
              type="password"
              value={password}
              onChange={e => setPassword(e.target.value)}
              autoComplete={mode === 'login' ? 'current-password' : 'new-password'}
            />
            {mode === 'register' && (
              <div style={{ fontSize: 11, color: '#9ca3af', marginTop: 4 }}>{t('passwordHint')}</div>
            )}
          </div>

          {mode === 'register' && (
            <div style={{ marginBottom: 16 }}>
              <label style={{ display: 'block', fontSize: 13, color: '#475467', marginBottom: 6 }}>
                {t('confirmPassword')}
              </label>
              <input
                style={inputStyle}
                type="password"
                value={confirm}
                onChange={e => setConfirm(e.target.value)}
                autoComplete="new-password"
              />
            </div>
          )}

          {error && (
            <div style={{ color: '#dc2626', fontSize: 13, marginBottom: 12 }}>{error}</div>
          )}

          <button type="submit" style={btnStyle} disabled={loading}>
            {loading ? '...' : (mode === 'login' ? t('loginBtn') : t('registerBtn'))}
          </button>
        </form>
      </div>
    </div>
  )
}
```
